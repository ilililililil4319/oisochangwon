```bash
git clone https://github.com/jojunsu98/oiso_changwon && cd oiso_changwon
pip install --break-system-packages -r requirements.txt qrcode && python -m unittest discover -s tests/ai_test   # 127개 통과
export NODE_PATH=$(npm root -g); npm i -g docx   # Word 변환용(없을 때)
bash handoff/tools/regen_all.sh   # md 고친 뒤 PDF·Word 다시 만들기(로고·QR 자동)
```
